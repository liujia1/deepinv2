# 16.7 动手感受一下：用 numpy/skimage 复现"角度越少，重建越糊"

> **目标**：用极度注释的 numpy + skimage 代码，走完 CT 逆问题的核心现象——给定同一张图，投影角度越少，FBP/反投影/伪逆重建越糊。这就是"不适定"的肉眼证据。代码可选运行：有 `skimage` 就用它；没有也能用我们手写的 numpy 版 `radon`/`iradon` 兜底。

**因果主线回顾**：正向模型 $y=Ax$（16.1）→ 角度不全导致 $A$ 病态（16.2）→ 反投影不是逆、FBP 靠斜坡滤波补救（16.1.4）→ 角度少则信息缺口大、重建糊。下面把这条线跑出来。

```python
# ============================================================
# 16.7  CT 不适定性的可运行演示
# 库：numpy / scipy / matplotlib / skimage（skimage 可选，有则用，无则退回手写）
# ============================================================
import numpy as np
import matplotlib.pyplot as plt

# ---- 1. 造一张测试图（用 skimage 自带的 Shepp-Logan 幻影）----
try:
    from skimage.data import shepp_logan_phantom
    from skimage.transform import resize
    HAVE_SKIMAGE = True
except Exception:
    HAVE_SKIMAGE = False

N = 128  # 图像边长（像素）

if HAVE_SKIMAGE:
    # 新版 skimage 的 shepp_logan_phantom() 不接受尺寸位置参数，先取默认再缩放
    ph = shepp_logan_phantom()
    x_true = resize(ph, (N, N), anti_aliasing=True).astype(np.float64)
    x_true = (x_true - x_true.min()) / (x_true.max() - x_true.min())  # 归一到 [0,1]
else:
    # 手写一个简易幻影：三个高斯团（skimage 不可用时兜底）
    yy, xx = np.mgrid[0:N, 0:N]
    def gauss(cx, cy, s):
        return np.exp(-((xx-cx)**2 + (yy-cy)**2) / (2*s*s))
    x_true = 0.6*gauss(N*0.35, N*0.4, N*0.12) + 0.9*gauss(N*0.65, N*0.6, N*0.09) + 0.5*gauss(N*0.5, N*0.75, N*0.06)
    x_true = (x_true - x_true.min()) / (x_true.max() - x_true.min())

# ---- 2. 正向模型：Radon 变换（投影 -> sinogram）----
# 我们优先用 skimage.transform.radon / iradon；否则用手写的 numpy 版。
if HAVE_SKIMAGE:
    from skimage.transform import radon, iradon

    def forward(x, theta):
        # theta: 角度数组(度)。返回 sinogram，形状 (探测器数, 角度数)
        return radon(x, theta=theta, circle=False)

    def fbp_recon(sinogram, theta):
        # 滤波反投影：skimage 的 iradon 默认带 ramp 滤波
        return iradon(sinogram, theta=theta, circle=False, filter_name='ramp')

    def backproj_recon(sinogram, theta):
        # 纯反投影（无滤波）：用 filter_name=None 得到模糊版本
        return iradon(sinogram, theta=theta, circle=False, filter_name=None)
else:
    # ---- 手写 numpy 版 Radon / 反投影（仅用于 skimage 缺失时兜底）----
    def _line_integral(x, angle_deg, n_det=N):
        # 对一个角度，做 n_det 条平行射线的线积分
        a = np.deg2rad(angle_deg)
        ca, sa = np.cos(a), np.sin(a)
        # 探测器坐标 t ∈ [-R, R]，R = N/√2
        R = N / np.sqrt(2.0)
        ts = np.linspace(-R, R, n_det)
        proj = np.zeros(n_det)
        for i, t in enumerate(ts):
            # 射线: x*ca + y*sa = t  -> 沿该方向采样图像
            # 用网格近似：构造该射线方向的坐标系，把图像重采样到"沿射线"的线积分
            # 简化做法：对每个像素，计算它到射线的带符号距离，距离<0.5则计入
            dist = np.abs(xx_ * ca + yy_ * sa - t)  # xx_, yy_ 为全局网格
            mask = dist < 0.5
            proj[i] = x[mask].sum()  # 近似线积分（像素贡献和）
        return proj

    yy_, xx_ = np.mgrid[0:N, 0:N]
    def forward(x, theta):
        return np.stack([_line_integral(x, th) for th in theta], axis=1)
    def backproj_recon(sinogram, theta):
        out = np.zeros_like(x_true)
        R = N / np.sqrt(2.0)
        ts = np.linspace(-R, R, sinogram.shape[0])
        for k, th in enumerate(theta):
            a = np.deg2rad(th); ca, sa = np.cos(a), np.sin(a)
            for i, t in enumerate(ts):
                dist = np.abs(xx_ * ca + yy_ * sa - t)
                out[dist < 0.5] += sinogram[i, k]
        return out
    def fbp_recon(sinogram, theta):
        # 手写版只做反投影(无精确 ramp 滤波)，此处回退到反投影以保可运行
        return backproj_recon(sinogram, theta)

# ---- 3. 试三组角度数：全角(180) / 稀疏(30) / 极少(12) ----
angles_list = {
    'full (180)': np.linspace(0, 180, 180, endpoint=False),
    'sparse (30)': np.linspace(0, 180, 30, endpoint=False),
    'very sparse (12)': np.linspace(0, 180, 12, endpoint=False),
}

# 计算一个粗略的 PSNR 评价（峰值信噪比，越大越清晰）
def psnr(a, b):
    mse = np.mean((a - b) ** 2)
    if mse <= 0:
        return float('inf')
    return 10 * np.log10(1.0 / mse)  # x_true 归一到 [0,1]，data_range=1

# ---- 4. 跑 reconstruction 并打印 PSNR ----
print("=== CT 不适定性演示：角度越少 -> 重建越糊 ===")
for name, theta in angles_list.items():
    sino = forward(x_true, theta)
    rec_fbp = fbp_recon(sino, theta)
    rec_bp = backproj_recon(sino, theta)
    p_fbp = psnr(x_true, np.clip(rec_fbp, 0, 1))
    p_bp = psnr(x_true, np.clip(rec_bp, 0, 1))
    print(f"[{name:>16}] 角度数={len(theta):3d}  FBP PSNR={p_fbp:5.1f} dB  反投影 PSNR={p_bp:5.1f} dB")

# ---- 5. 画图：真值 + 三组 FBP 重建，肉眼看"糊" ----
fig, axes = plt.subplots(1, 4, figsize=(16, 4))
axes[0].imshow(x_true, cmap='gray'); axes[0].set_title('真值 x_true'); axes[0].axis('off')
for ax, (name, theta) in zip(axes[1:], angles_list.items()):
    sino = forward(x_true, theta)
    rec = fbp_recon(sino, theta)
    rec = np.clip(rec, 0, 1)
    ax.imshow(rec, cmap='gray')
    ax.set_title(f"{name}\nFBP PSNR={psnr(x_true, rec):.1f} dB")
    ax.axis('off')
plt.suptitle("同一张图，投影角度越少 -> 重建越糊（不适定现象）", fontsize=13)
plt.tight_layout()
plt.show()

# ---- 6. 额外：反投影(无滤波) vs FBP，证明"反投影≠逆" ----
theta = angles_list['full (180)']
sino = forward(x_true, theta)
rec_bp = np.clip(backproj_recon(sino, theta), 0, 1)
rec_fbp = np.clip(fbp_recon(sino, theta), 0, 1)
fig, axes = plt.subplots(1, 3, figsize=(12, 4))
axes[0].imshow(x_true, cmap='gray'); axes[0].set_title('真值'); axes[0].axis('off')
axes[1].imshow(rec_bp, cmap='gray'); axes[1].set_title(f'反投影(无滤波)\nPSNR={psnr(x_true, rec_bp):.1f} dB'); axes[1].axis('off')
axes[2].imshow(rec_fbp, cmap='gray'); axes[2].set_title(f'FBP(有 ramp 滤波)\nPSNR={psnr(x_true, rec_fbp):.1f} dB'); axes[2].axis('off')
plt.suptitle("反投影 ≠ 逆投影：ramp 滤波把模糊变清晰（呼应 16.1.4）", fontsize=13)
plt.tight_layout()
plt.show()
```

**你会看到什么**：
- 三组角度（180 / 30 / 12）下，FBP 的 PSNR 随角度数递减——角度越少，条纹伪影、混叠越明显，肉眼可见"糊"。这正是第 1 章逆问题 + 第 16.2 节"不适定"的直观证据。
- 第二组图：全角下反投影（无滤波）明显比 FBP 模糊——证明"反投影不是 Radon 的逆，FBP 靠 ramp 滤波器 $|\omega|$ 补偿平滑"（16.1.4）。

**延伸（可选动手）**：把 `forward` 的输出加上高斯噪声 `sino += 0.02*np.random.randn(*sino.shape)`，再重建，会发现 FBP 对噪声敏感（ramp 放大高频噪声，16.1.4 已讲）——这就自然引出 16.2 的正则化与 16.5 的扩散先验为何必要。
